# Start the server
node main.js -h 127.0.0.1 -p 3000 -c ./cache

# Missing required options
node main.js
node main.js -h 127.0.0.1 -p 3000
node main.js -h 127.0.0.1 -p abc -c ./cache

# Malformed paths
curl -i http://127.0.0.1:3000/abc
curl -i --path-as-is http://127.0.0.1:3000/../etc/passwd

# GET an image that is already in the cache (put the file there manually first)
curl -s https://http.cat/418 -o cache/418.jpg
curl -i http://127.0.0.1:3000/418 -o /dev/null -D -
curl -s http://127.0.0.1:3000/418 -o out.jpg && cmp out.jpg cache/418.jpg

# GET an image that is not cached
curl -i http://127.0.0.1:3000/404

# GET an image that is not cached: fetched from http.cat, then served from the cache
rm -rf cache
curl -i http://127.0.0.1:3000/418 -o out1.jpg -D -
curl -i http://127.0.0.1:3000/418 -o out2.jpg -D -
cmp out1.jpg out2.jpg
ls -l cache

# Code accepted by the proxy but missing on http.cat -> 404, nothing is cached
curl -i http://127.0.0.1:3000/299
ls cache

# Out of range code -> 400
curl -i http://127.0.0.1:3000/999

# http.cat unreachable (server started in a network namespace without network access)
unshare -rn bash -c 'node main.js -h 127.0.0.1 -p 3000 -c ./cache & sleep 1; curl -i http://127.0.0.1:3000/200'

# PUT: store a custom image, then read it back (served from the cache, byte for byte)
curl -i -X PUT --data-binary @samples/custom-image.jpg -H 'Content-Type: image/jpeg' http://127.0.0.1:3000/418
cmp cache/418.jpg samples/custom-image.jpg
curl -i http://127.0.0.1:3000/418 -o out3.jpg -D -
cmp out3.jpg samples/custom-image.jpg

# PUT with an empty body -> 400
curl -i -X PUT http://127.0.0.1:3000/200

# DELETE: remove a cached image, then delete it again -> 404, then GET fetches it from http.cat again
curl -i -X DELETE http://127.0.0.1:3000/418
ls cache
curl -i -X DELETE http://127.0.0.1:3000/418
curl -i http://127.0.0.1:3000/418 -o /dev/null -D -

# Unsupported methods -> 405 with an Allow header
curl -i -X POST http://127.0.0.1:3000/418
curl -i -X PATCH http://127.0.0.1:3000/418
curl -I http://127.0.0.1:3000/418

# PUT with a body over 10 MB -> 413, nothing is stored
head -c 11000000 /dev/urandom > /tmp/big.bin
curl -i -X PUT --data-binary @/tmp/big.bin http://127.0.0.1:3000/200
ls cache

# Five parallel GETs for an image that is not cached yet: one download, identical results
rm -rf cache
for i in 1 2 3 4 5; do curl -s -D - -o par$i.jpg http://127.0.0.1:3000/203 | grep -i source & done; wait
md5sum par*.jpg
ls cache

# Cache directory removed while the server is running: it is recreated on the next write
rm -rf cache
curl -i http://127.0.0.1:3000/418 -o /dev/null -D -
ls cache
